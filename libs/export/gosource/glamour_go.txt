package main

// glamour.go renders the "Markdown" component with Glamour. The text in the
// design uses | for a new line.

import (
	"strings"

	"charm.land/glamour/v2"
)

func init() { extra["glamour.markdown"] = markdownPart }

func markdownPart(c placed, p props) *part {
	style := p.str("style")
	switch style {
	case "dark", "light", "pink", "notty":
	default:
		style = "dark"
	}
	r, err := glamour.NewTermRenderer(glamour.WithStandardStyle(style), glamour.WithWordWrap(max(c.W, 10)))
	if err != nil {
		return staticPart(box(c.W, c.H, "normal", "240", "markdown", ""))
	}
	out, err := r.Render(strings.ReplaceAll(p.str("markdown"), "|", "\n"))
	if err != nil {
		return staticPart(box(c.W, c.H, "normal", "240", "markdown", ""))
	}
	return staticPart(strings.Trim(out, "\n"))
}
