package main

// bigtext.go draws the "Big text" component with go-figure. It is a separate
// file because it needs a module of its own; runtime.go does not know about it.

import (
	"strings"

	figure "github.com/common-nighthawk/go-figure"
)

func init() { extra["community.bigtext"] = bigTextPart }

// figletFonts are the fonts Cuppa offers. go-figure ends the whole program on a
// font it does not know, so any other name falls back to "standard".
var figletFonts = map[string]bool{
	"standard": true, "big": true, "small": true, "mini": true, "slant": true,
	"shadow": true, "doom": true, "banner": true, "block": true, "lean": true,
}

func bigTextPart(c placed, p props) *part {
	font := p.str("font")
	if !figletFonts[font] {
		font = "standard"
	}
	text := strings.TrimSpace(p.str("text"))
	var rows []string
	if text != "" {
		rows = figure.NewFigure(text, font, false).Slicify()
	}
	var lines []string
	for i, row := range rows {
		if i >= c.H {
			break
		}
		lines = append(lines, fg(p.str("color")).MaxWidth(c.W).Render(row))
	}
	return staticPart(strings.Join(lines, "\n"))
}
