package main

// huh.go runs the Huh fields of the design: each field is a Huh form with one
// group, so it behaves as it does in a Huh form (typing, arrows, Enter).

import (
	"strconv"

	tea "charm.land/bubbletea/v2"
	"charm.land/huh/v2"
)

func init() {
	for _, kind := range []string{"huh.input", "huh.text", "huh.select", "huh.multiselect", "huh.confirm", "huh.note", "huh.filepicker"} {
		extra[kind] = huhFieldPart
	}
}

func huhFieldPart(c placed, p props) (out *part) {
	defer func() {
		if r := recover(); r != nil {
			out = staticPart(box(c.W, c.H, "normal", "240", c.Kind, ""))
		}
	}()
	title, description := p.str("title"), p.str("description")
	var field huh.Field
	switch c.Kind {
	case "huh.input":
		value := p.str("value")
		field = huh.NewInput().Title(title).Description(description).Placeholder(p.str("placeholder")).Value(&value)
	case "huh.text":
		value := p.str("value")
		field = huh.NewText().Title(title).Description(description).Placeholder(p.str("placeholder")).Lines(max(c.H-3, 1)).Value(&value)
	case "huh.select":
		options := split(p.str("options"), ",")
		chosen := ""
		if i := p.integer("selected", 0); i >= 0 && i < len(options) {
			chosen = options[i]
		}
		field = huh.NewSelect[string]().Title(title).Description(description).
			Options(huh.NewOptions(options...)...).Height(max(c.H-1, 3)).Value(&chosen)
	case "huh.multiselect":
		checked := map[int]bool{}
		for _, s := range split(p.str("checked"), ",") {
			if n, err := strconv.Atoi(s); err == nil {
				checked[n] = true
			}
		}
		var options []huh.Option[string]
		for i, o := range split(p.str("options"), ",") {
			options = append(options, huh.NewOption(o, o).Selected(checked[i]))
		}
		var chosen []string
		field = huh.NewMultiSelect[string]().Title(title).Description(description).Options(options...).
			Height(max(c.H-1, 3)).Value(&chosen)
	case "huh.confirm":
		yes := p.flag("value")
		field = huh.NewConfirm().Title(title).Description(description).
			Affirmative(orDefault(p.str("affirmative"), "Yes")).Negative(orDefault(p.str("negative"), "No")).Value(&yes)
	case "huh.note":
		field = huh.NewNote().Title(title).Description(p.str("body"))
	case "huh.filepicker":
		picked := ""
		field = huh.NewFilePicker().Title(title).Description(description).CurrentDirectory(".").
			Picking(true).Height(max(c.H-2, 3)).Value(&picked)
	}
	form := huh.NewForm(huh.NewGroup(field)).WithShowHelp(false).WithWidth(c.W).WithHeight(c.H)
	return &part{
		keys:  true,
		start: form.Init(),
		update: func(msg tea.Msg) tea.Cmd {
			m, cmd := form.Update(msg)
			if f, ok := m.(*huh.Form); ok {
				form = f
			}
			return cmd
		},
		view: func() string { return form.View() },
	}
}
