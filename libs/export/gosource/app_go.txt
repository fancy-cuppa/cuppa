package main

import (
	"fmt"
	"os"

	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

// model is the whole program: the components of layout.go, running, and which
// of them has the keyboard.
type model struct {
	parts []*part
	// focus is the index of the component that reads the keyboard, or -1.
	focus int
}

func newModel() model {
	m := model{focus: -1}
	for _, c := range layout {
		m.parts = append(m.parts, build(c))
	}
	return m
}

func (m model) Init() tea.Cmd {
	var starts []tea.Cmd
	for _, p := range m.parts {
		starts = append(starts, p.start)
	}
	return tea.Batch(starts...)
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyPressMsg:
		switch msg.String() {
		case "ctrl+c", "esc":
			return m, tea.Quit
		case "tab":
			return m, m.cycle(1)
		case "shift+tab":
			return m, m.cycle(-1)
		}
		if m.focus >= 0 {
			return m, m.parts[m.focus].update(msg)
		}
		return m, nil
	case tea.MouseClickMsg:
		return m, m.focusAt(msg.X, msg.Y)
	case tea.MouseWheelMsg:
		if i := m.at(msg.X, msg.Y); i >= 0 {
			code := tea.KeyDown
			if msg.Button == tea.MouseWheelUp {
				code = tea.KeyUp
			}
			return m, m.parts[i].update(tea.KeyPressMsg{Code: code})
		}
		return m, nil
	}
	// Everything else (timer ticks, cursor blinks) goes to every component; each
	// one ignores what is not its own.
	var cmds []tea.Cmd
	for _, p := range m.parts {
		cmds = append(cmds, p.update(msg))
	}
	return m, tea.Batch(cmds...)
}

// cycle moves the keyboard to the next (or previous) component that reads it.
func (m *model) cycle(step int) tea.Cmd {
	n := len(m.parts)
	for i := 1; i <= n; i++ {
		next := ((m.focus+step*i)%n + n) % n
		if m.parts[next].keys {
			return m.setFocus(next)
		}
	}
	return nil
}

func (m *model) setFocus(i int) tea.Cmd {
	if i == m.focus {
		return nil
	}
	var cmds []tea.Cmd
	if m.focus >= 0 && m.parts[m.focus].focus != nil {
		cmds = append(cmds, m.parts[m.focus].focus(false))
	}
	m.focus = i
	if i >= 0 && m.parts[i].focus != nil {
		cmds = append(cmds, m.parts[i].focus(true))
	}
	return tea.Batch(cmds...)
}

// at is the top-most component under the cell, or -1.
func (m model) at(x, y int) int {
	for i := len(layout) - 1; i >= 0; i-- {
		c := layout[i]
		if x >= c.X && x < c.X+c.W && y >= c.Y && y < c.Y+c.H {
			return i
		}
	}
	return -1
}

func (m *model) focusAt(x, y int) tea.Cmd {
	i := m.at(x, y)
	if i < 0 || !m.parts[i].keys {
		i = -1
	}
	return m.setFocus(i)
}

func (m model) View() tea.View {
	layers := []*lipgloss.Layer{lipgloss.NewLayer(backdrop()).Z(0)}
	for i, c := range layout {
		layers = append(layers, lipgloss.NewLayer(m.parts[i].view()).X(c.X).Y(c.Y).Z(i+1))
	}
	v := tea.NewView(lipgloss.NewCompositor(layers...).Render())
	v.AltScreen = true
	v.MouseMode = tea.MouseModeCellMotion
	return v
}

// backdrop is the canvas itself: its size and background colour.
func backdrop() string {
	row := make([]byte, canvasW)
	for i := range row {
		row[i] = ' '
	}
	line := string(row)
	if background != "" {
		line = lipgloss.NewStyle().Background(lipgloss.Color(background)).Render(line)
	}
	lines := make([]string, canvasH)
	for i := range lines {
		lines[i] = line
	}
	return joinLines(lines)
}

func joinLines(lines []string) string {
	out := ""
	for i, l := range lines {
		if i > 0 {
			out += "\n"
		}
		out += l
	}
	return out
}

func main() {
	if _, err := tea.NewProgram(newModel()).Run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
