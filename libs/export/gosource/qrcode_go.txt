package main

// qrcode.go draws the "QR code" component with go-qrcode, two modules to a row
// in half blocks, dark on a light ground so that it scans on a dark terminal.

import (
	"strings"

	"charm.land/lipgloss/v2"
	qrcode "github.com/skip2/go-qrcode"
)

func init() { extra["community.qrcode"] = qrCodePart }

func qrCodePart(c placed, p props) *part {
	levels := map[string]qrcode.RecoveryLevel{
		"low": qrcode.Low, "medium": qrcode.Medium, "high": qrcode.High, "highest": qrcode.Highest,
	}
	level, ok := levels[p.str("level")]
	if !ok {
		level = qrcode.Medium
	}
	content := p.str("content")
	code, err := qrcode.New(content, level)
	if content == "" || err != nil {
		return staticPart(box(c.W, c.H, "normal", "240", "QR code", ""))
	}
	bits := code.Bitmap()
	dark := func(x, y int) bool { return y < len(bits) && x < len(bits[y]) && bits[y][x] }
	style := lipgloss.NewStyle().Foreground(lipgloss.Color("0")).Background(lipgloss.Color("255"))
	var lines []string
	for y := 0; y < c.H; y++ {
		var row strings.Builder
		for x := 0; x < c.W; x++ {
			top, bottom := dark(x, 2*y), dark(x, 2*y+1)
			switch {
			case top && bottom:
				row.WriteString("█")
			case top:
				row.WriteString("▀")
			case bottom:
				row.WriteString("▄")
			default:
				row.WriteString(" ")
			}
		}
		lines = append(lines, style.Render(row.String()))
	}
	return staticPart(strings.Join(lines, "\n"))
}
