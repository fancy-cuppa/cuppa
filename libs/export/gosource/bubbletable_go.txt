package main

// bubbletable.go runs the "Bubble table" component with Evertras/bubble-table:
// a table you can move through with the arrow keys, with a footer line.

import (
	"strconv"

	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
	"github.com/evertras/bubble-table/table"
)

func init() { extra["community.bubbletable"] = bubbleTablePart }

func bubbleTablePart(c placed, p props) (out *part) {
	defer func() {
		if r := recover(); r != nil {
			out = staticPart(box(c.W, c.H, "normal", "240", "table", ""))
		}
	}()
	names := split(p.str("columns"), ",")
	each := max((c.W-len(names)-1)/max(len(names), 1)-2, 3)
	var columns []table.Column
	for i, n := range names {
		columns = append(columns, table.NewColumn("c"+strconv.Itoa(i), n, each))
	}
	var rows []table.Row
	for _, line := range split(p.str("rows"), ";") {
		data := table.RowData{}
		for i, cell := range split(line, ",") {
			data["c"+strconv.Itoa(i)] = cell
		}
		rows = append(rows, table.NewRow(data))
	}
	m := table.New(columns).WithRows(rows).Focused(true).
		HighlightStyle(lipgloss.NewStyle().Background(lipgloss.Color(orDefault(p.str("color"), "212"))).Foreground(lipgloss.Color("0")))
	if footer := p.str("footer"); footer != "" {
		m = m.WithStaticFooter(footer)
	}
	return &part{
		keys:   true,
		update: func(msg tea.Msg) tea.Cmd { var cmd tea.Cmd; m, cmd = m.Update(msg); return cmd },
		view:   func() string { return m.View() },
	}
}
