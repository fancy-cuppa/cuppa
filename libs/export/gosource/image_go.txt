package main

// image.go draws the "Image" component: it reads the file named by the path
// property (relative to where the program runs) and shows it with half blocks,
// two pixels to a cell. It uses only the standard library. For Kitty, Sixel or
// iTerm2 graphics, swap this for a library such as go-termimg.

import (
	"fmt"
	"image"
	_ "image/gif"
	_ "image/jpeg"
	_ "image/png"
	"os"
	"strings"

	"charm.land/lipgloss/v2"
)

func init() { extra["community.image"] = imagePart }

func imagePart(c placed, p props) *part {
	path := p.str("path")
	f, err := os.Open(path)
	if err != nil {
		return staticPart(box(c.W, c.H, "normal", "240", path, ""))
	}
	defer f.Close()
	img, _, err := image.Decode(f)
	if err != nil || img.Bounds().Dx() == 0 || img.Bounds().Dy() == 0 {
		return staticPart(box(c.W, c.H, "normal", "240", path, ""))
	}
	return staticPart(halfBlocks(img, c.W, c.H, p.str("fit")))
}

// halfBlocks scales img into w columns and h rows (2h pixels tall). "fit"
// shows it whole, "fill" crops it to cover, "stretch" distorts it to the box.
func halfBlocks(img image.Image, w, h int, fit string) string {
	b := img.Bounds()
	iw, ih := float64(b.Dx()), float64(b.Dy())
	pw, ph := float64(w), float64(2*h)
	var scaleX, scaleY float64
	switch fit {
	case "stretch":
		scaleX, scaleY = pw/iw, ph/ih
	case "fill":
		scaleX = max(pw/iw, ph/ih)
		scaleY = scaleX
	default:
		scaleX = min(pw/iw, ph/ih)
		scaleY = scaleX
	}
	dw, dh := iw*scaleX, ih*scaleY
	offX, offY := (pw-dw)/2, (ph-dh)/2
	pixel := func(x, y int) (string, bool) {
		sx, sy := (float64(x)-offX)/scaleX, (float64(y)-offY)/scaleY
		if sx < 0 || sy < 0 || sx >= iw || sy >= ih {
			return "", false
		}
		r, g, bl, _ := img.At(b.Min.X+int(sx), b.Min.Y+int(sy)).RGBA()
		return fmt.Sprintf("#%02x%02x%02x", r>>8, g>>8, bl>>8), true
	}
	var lines []string
	for y := 0; y < h; y++ {
		var row strings.Builder
		for x := 0; x < w; x++ {
			top, okTop := pixel(x, 2*y)
			bottom, okBottom := pixel(x, 2*y+1)
			switch {
			case okTop && okBottom:
				row.WriteString(lipgloss.NewStyle().Foreground(lipgloss.Color(top)).Background(lipgloss.Color(bottom)).Render("▀"))
			case okTop:
				row.WriteString(lipgloss.NewStyle().Foreground(lipgloss.Color(top)).Render("▀"))
			case okBottom:
				row.WriteString(lipgloss.NewStyle().Foreground(lipgloss.Color(bottom)).Render("▄"))
			default:
				row.WriteString(" ")
			}
		}
		lines = append(lines, row.String())
	}
	return strings.Join(lines, "\n")
}
