package main

// charts.go draws the chart components with ntcharts. The values in the design
// are comma separated numbers; a chart that cannot be drawn at the size it was
// given falls back to an empty frame.

import (
	"strconv"
	"strings"
	"time"

	"github.com/NimbleMarkets/ntcharts/v2/barchart"
	"github.com/NimbleMarkets/ntcharts/v2/canvas"
	"github.com/NimbleMarkets/ntcharts/v2/heatmap"
	"github.com/NimbleMarkets/ntcharts/v2/linechart"
	"github.com/NimbleMarkets/ntcharts/v2/linechart/streamlinechart"
	"github.com/NimbleMarkets/ntcharts/v2/linechart/timeserieslinechart"
	"github.com/NimbleMarkets/ntcharts/v2/sparkline"
)

func init() {
	extra["ntcharts.sparkline"] = chartPart(drawSparkline)
	extra["ntcharts.barchart"] = chartPart(drawBarChart)
	extra["ntcharts.linechart"] = chartPart(drawLineChart)
	extra["ntcharts.streamline"] = chartPart(drawStreamline)
	extra["ntcharts.timeseries"] = chartPart(drawTimeSeries)
	extra["ntcharts.heatmap"] = chartPart(drawHeatMap)
	extra["ntcharts.canvas"] = chartPart(drawChartCanvas)
}

// chartPart makes a static component from a drawing function, and falls back to
// a frame when the chart library cannot draw at that size.
func chartPart(draw func(c placed, p props) string) func(c placed, p props) *part {
	return func(c placed, p props) (out *part) {
		defer func() {
			if r := recover(); r != nil {
				out = staticPart(box(c.W, c.H, "normal", "240", strings.TrimPrefix(c.Kind, "ntcharts."), ""))
			}
		}()
		return staticPart(draw(c, p))
	}
}

func numbersOf(p props, key string) []float64 {
	var out []float64
	for _, s := range split(p.str(key), ",") {
		if v, err := strconv.ParseFloat(s, 64); err == nil {
			out = append(out, v)
		}
	}
	return out
}

func extentOf(values []float64) (lo, hi float64) {
	if len(values) == 0 {
		return 0, 1
	}
	lo, hi = values[0], values[0]
	for _, v := range values {
		lo, hi = min(lo, v), max(hi, v)
	}
	if hi == lo {
		hi = lo + 1
	}
	return lo, hi
}

func drawSparkline(c placed, p props) string {
	m := sparkline.New(c.W, c.H, sparkline.WithStyle(fg(p.str("color"))))
	m.PushAll(numbersOf(p, "values"))
	m.Draw()
	return m.View()
}

func drawBarChart(c placed, p props) string {
	values := numbersOf(p, "values")
	labels := split(p.str("labels"), ",")
	var data []barchart.BarData
	for i, v := range values {
		label := ""
		if i < len(labels) {
			label = labels[i]
		}
		data = append(data, barchart.BarData{
			Label:  label,
			Values: []barchart.BarValue{{Name: label, Value: v, Style: fg(p.str("color"))}},
		})
	}
	m := barchart.New(c.W, c.H, barchart.WithDataSet(data))
	m.Draw()
	return m.View()
}

func drawLineChart(c placed, p props) string {
	values := numbersOf(p, "values")
	lo, hi := extentOf(values)
	m := linechart.New(c.W, c.H, 0, float64(max(len(values)-1, 1)), lo, hi)
	if p.flag("axes") {
		m.DrawXYAxisAndLabel()
	}
	for i := 1; i < len(values); i++ {
		m.DrawBrailleLineWithStyle(
			canvas.Float64Point{X: float64(i - 1), Y: values[i-1]},
			canvas.Float64Point{X: float64(i), Y: values[i]},
			fg(p.str("color")))
	}
	return m.View()
}

func drawStreamline(c placed, p props) string {
	m := streamlinechart.New(c.W, c.H, streamlinechart.WithStyles(0, fg(p.str("color"))))
	for _, v := range numbersOf(p, "values") {
		m.Push(v)
	}
	m.Draw()
	return m.View()
}

func drawTimeSeries(c placed, p props) string {
	values := numbersOf(p, "values")
	start := time.Date(2026, 1, 1, 0, 0, 0, 0, time.UTC)
	end := start.Add(time.Duration(max(len(values)-1, 1)) * time.Hour)
	m := timeserieslinechart.New(c.W, c.H,
		timeserieslinechart.WithStyle(fg(p.str("color"))),
		timeserieslinechart.WithTimeRange(start, end),
		timeserieslinechart.WithXLabelFormatter(timeserieslinechart.HourTimeLabelFormatter()))
	for i, v := range values {
		m.Push(timeserieslinechart.TimePoint{Time: start.Add(time.Duration(i) * time.Hour), Value: v})
	}
	m.DrawBraille()
	return m.View()
}

func drawHeatMap(c placed, p props) string {
	seed := p.integer("seed", 0)
	m := heatmap.New(c.W, c.H)
	var points []heatmap.HeatPoint
	for y := 0; y < c.H; y++ {
		for x := 0; x < c.W; x++ {
			points = append(points, heatmap.NewHeatPointInt(x, y, float64((x*7+y*13+seed*31+(x*y)%5*3)%16)))
		}
	}
	m.PushAll(points)
	m.Draw()
	return m.View()
}

func drawChartCanvas(c placed, p props) string {
	m := canvas.New(c.W, c.H)
	if title := strings.TrimSpace(p.str("title")); title != "" && c.W > 6 {
		m.SetRunesWithStyle(canvas.Point{X: max(c.W-len([]rune(title))-1, 0), Y: 0}, []rune(title), fg(p.str("color")))
	}
	return m.View()
}
